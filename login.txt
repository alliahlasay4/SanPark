    // Password Strength Calculation
    function calculatePasswordStrength(pass) {
      const bar = document.getElementById('strength-bar');
      const label = document.getElementById('strength-label');
      const reqLen = document.getElementById('req-len');
      const reqNum = document.getElementById('req-num');
      const reqSym = document.getElementById('req-sym');

      const hasLen = pass.length >= 8;
      const hasNum = /\d/.test(pass);
      const hasSym = /[!@#$%^&*(),.?":{}|<>]/.test(pass);

      // Update badge checks
      updateRequirement(reqLen, hasLen, '8+ chars');
      updateRequirement(reqNum, hasNum, '1 number');
      updateRequirement(reqSym, hasSym, '1 symbol');

      let score = 0;
      if (hasLen) score++;
      if (hasNum) score++;
      if (hasSym) score++;

      if (score <= 1) {
        bar.className = "h-full bg-status-red transition-all duration-300 w-1/3";
        label.className = "text-[11px] font-semibold text-status-red";
        label.innerText = "Weak Password";
      } else if (score === 2) {
        bar.className = "h-full bg-status-amber transition-all duration-300 w-2/3";
        label.className = "text-[11px] font-semibold text-status-amber";
        label.innerText = "Medium Strength";
      } else {
        bar.className = "h-full bg-status-green transition-all duration-300 w-full";
        label.className = "text-[11px] font-semibold text-status-green";
        label.innerText = "Strong Password (Encrypted)";
      }
    }

        el.className = "flex items-center gap-1 text-white/30";
        el.innerHTML = `<i data-lucide="circle" class="w-2.5 h-2.5"></i> ${text}`;
      }
      lucide.createIcons();
    }

      switchTab('otp');
    }

    // OAuth Social Simulation
    function simulateOAuth(provider) {
      showCustomToast(`${provider} Connect`, `Handshaking with ${provider} identity provider. Auto-filling profile credentials.`);
      setTimeout(() => {
        switchTab('otp');
      }, 800);
    }

    // Forgot Password Simulation
    function triggerForgotPassword() {
      const email = document.getElementById('login-identifier').value;
      showCustomToast("Password Recovery Dispatched", `A secure reset link has been dispatched to ${email || 'your registered address'}.`);
    }

    // OTP 6-Digit Auto-Advancing Boxes
    const otpBoxes = document.querySelectorAll('.otp-box');
    otpBoxes.forEach((box, index) => {
      box.addEventListener('input', (e) => {
        const val = e.target.value;
        if (val.length === 1 && index < otpBoxes.length - 1) {
          otpBoxes[index + 1].focus();
        }
      });

      box.addEventListener('keydown', (e) => {
        if (e.key === 'Backspace' && !e.target.value && index > 0) {
          otpBoxes[index - 1].focus();
        }
      });

      box.addEventListener('paste', (e) => {
        e.preventDefault();
        const pasteData = (e.clipboardData || window.clipboardData).getData('text').trim();
        if (/^\d+$/.test(pasteData)) {
          const digits = pasteData.slice(0, 6).split('');
          digits.forEach((d, i) => {
            if (otpBoxes[i]) otpBoxes[i].value = d;
          });
          if (digits.length >= 6) {
            verifySubmittedOtp();
          } else if (otpBoxes[digits.length]) {
            otpBoxes[digits.length].focus();
          }
        }
      });
    });

    function fillCorrectDemoOtp() {
      const demo = ['1', '2', '3', '4', '5', '6'];
      otpBoxes.forEach((box, i) => {
        box.value = demo[i];
      });
      document.getElementById('otp-error-msg').classList.add('hidden');
      showCustomToast("OTP Auto-Filled", "Demo token 123456 loaded. Click 'Verify & Access Dashboard'.");
    }

    function verifySubmittedOtp() {
      let enteredCode = '';
      otpBoxes.forEach(b => enteredCode += b.value);

      const errorMsg = document.getElementById('otp-error-msg');
      const cardContainer = document.getElementById('otp-card-container');
      const successState = document.getElementById('otp-success-state');

      if (enteredCode === '123456') {
        errorMsg.classList.add('hidden');
        cardContainer.classList.add('hidden');
        successState.classList.remove('hidden');
        successState.classList.add('flex');
        lucide.createIcons();
        showCustomToast("Access Granted", "Welcome to SanPark Urban Hub. All gates unlocked.");
      } else {
        errorMsg.classList.remove('hidden');
        cardContainer.classList.add('animate-shake');
        setTimeout(() => {
          cardContainer.classList.remove('animate-shake');
        }, 500);
        showCustomToast("Verification Failed", "Entered token does not match. Demo code is 123456.", false);
      }
    }

    // Countdown Timer logic
    let secondsLeft = 48;
    const timerDisplay = document.getElementById('countdown-timer');
    const resendBtn = document.getElementById('resend-otp-btn');

    setInterval(() => {
      if (secondsLeft > 0) {
        secondsLeft--;
        const mins = String(Math.floor(secondsLeft / 60)).padStart(2, '0');
        const secs = String(secondsLeft % 60).padStart(2, '0');
        timerDisplay.innerText = `${mins}:${secs}`;
      } else {
        resendBtn.disabled = false;
        resendBtn.classList.remove('opacity-40', 'cursor-not-allowed');
      }
    }, 1000);

    function triggerResendOtp() {
      secondsLeft = 60;
      showCustomToast("New Token Dispatched", "A fresh 6-digit security code has been transmitted.");
      resendBtn.disabled = true;
    }
